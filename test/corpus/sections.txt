==================
Summary and all core sections
==================

Fetch a record.

Args:
  key: Record key.

Returns:
  The matching record.

Raises:
  KeyError: No record exists.

Attributes:
  cache: Cached records.

Yields:
  A record.

---

(document
  (text)
  (args_section
    header: (section_header)
    (parameter
      name: (identifier)
      description: (description
        (text))))
  (returns_section
    header: (section_header)
    (return_value
      description: (description
        (text))))
  (raises_section
    header: (section_header)
    (exception
      name: (exception_name)
      description: (description
        (text))))
  (attributes_section
    header: (section_header)
    (attribute
      name: (identifier)
      description: (description
        (text))))
  (yields_section
    header: (section_header)
    (return_value
      description: (description
        (text)))))
